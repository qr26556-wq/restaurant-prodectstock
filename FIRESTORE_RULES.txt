// جاؤ Firebase Console > Firestore Database > Rules اور یہ rules پیسٹ کر کے Publish کریں:
//
// PEHLE, YE ZAROORI HAI (warna neeche wale rules kaam nahi karenge):
// 1. Firebase Console > Authentication > Sign-in method > "Email/Password" ko Enable karein.
// 2. Firebase Console > Authentication > Users > "Add user" se ek account banayein
//    apne admin email (jo js/app.js mein ADMIN_EMAIL hai) aur password ke sath.
// Is ke baghair customer signup bhi fail hoga (auth/configuration-not-found error)
// aur admin login bhi kaam nahi karega.

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    function isAdmin() {
      // ADMIN_EMAIL yahan bhi wahi hona chahiye jo js/app.js mein hai.
      return request.auth != null && request.auth.token.email == 'qraza2376@gmail.com';
    }

    // Products — sab visitors read kar sakte hain (store browse karne ke
    // liye zaroori hai), lekin ab sirf admin account hi likh/edit/delete
    // kar sakta hai. Pehle "write: if true" tha — koi bhi browser console
    // se seedha Firestore mein product add/edit/delete kar sakta tha.
    match /products/{productId} {
      allow read: if true;
      allow write: if isAdmin();
    }

    // User liked products — Firebase Auth (email/password) use karta hai,
    // har user sirf apna data padh/likh sakta hai.
    match /users/{userId} {
      allow read: if request.auth.uid == userId;
      allow write: if request.auth.uid == userId;

      match /addresses/{addressId} {
        allow read, write: if request.auth != null && request.auth.uid == userId;
      }
    }

    // Orders — customer checkout par order create kar sakta hai (create
    // rehne diya hai "if true" taake bina login order de sakein), lekin
    // status update/delete ab sirf admin kar sakta hai.
    match /orders/{orderId} {
      allow read: if true;
      allow create: if true;
      allow update: if isAdmin() ||
        request.resource.data.diff(resource.data).affectedKeys().hasOnly(['cancelRequested','cancelRequestedAt','returnRequested','returnReason','returnRequestedAt']);
      allow delete: if isAdmin();
    }

    // Analytics summary (site visits, product views, add-to-cart counts) —
    // ek hi document mein counters store hote hain, koi personal data nahi.
    // NOTE (fix): trackEvent() in app.js writes these same counters from
    // anonymous visitors too, so a plain isAdmin()-only rule blocked every
    // visitor's write with "Missing or insufficient permissions". Ab sirf
    // known counter fields ko admin ke baghair bhi likhne diya gaya hai —
    // baaki sab kuch (naye fields, doc delete waghera) abhi bhi admin-only
    // hai. See firestore.rules at the project root — that's the file that
    // actually gets deployed; this .txt is kept only as a readable copy.
    match /analytics/{docId} {
      allow read: if true;
      allow write: if isAdmin();
      allow create: if request.resource.data.keys().hasOnly(
        ['lastUpdated','totalPageViews','totalProductViews','productViews','totalAddToCart','productAddToCart']);
      allow update: if request.resource.data.diff(resource.data).affectedKeys().hasOnly(
        ['lastUpdated','totalPageViews','totalProductViews','productViews','totalAddToCart','productAddToCart']);
    }

    // Homepage hero banners — sab dekh sakte hain, sirf admin add/edit/
    // delete kar sakta hai.
    match /banners/{bannerId} {
      allow read: if true;
      allow write: if isAdmin();
    }

    // Coupons — customers checkout par sirf ek specific code "get" kar
    // sakte hain (validate karne ke liye), lekin poori list nahi dekh
    // sakte (isse random discovery rukti hai). Order place karte waqt
    // sirf "usedCount" field update ho sakta hai — discount value ya
    // active/expiry customer badal nahi sakta. Add/delete sirf admin.
    match /coupons/{code} {
      allow get: if true;
      allow list: if isAdmin();
      allow create, delete: if isAdmin();
      allow update: if isAdmin() ||
        request.resource.data.diff(resource.data).affectedKeys().hasOnly(['usedCount']);
    }

    // Product reviews — koi bhi likh (create) sakta hai (login zaroori
    // nahi, jaise orders ke liye hai), lekin edit/delete sirf admin kar
    // sakta hai taake fake/spam reviews hataye ja sakein.
    match /reviews/{reviewId} {
      allow read: if true;
      allow create: if request.resource.data.rating is int
        && request.resource.data.rating >= 1
        && request.resource.data.rating <= 5
        && request.resource.data.productId is string;
      allow update, delete: if isAdmin();
    }
  }
}
